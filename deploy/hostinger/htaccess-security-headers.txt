# Security headers + caching for the MICO360 Tasks web app on Hostinger (LiteSpeed reads .htaccess).
# MERGE these lines into the existing .htaccess in the web root that serves the React build —
# do not replace that file, it may already contain Hostinger's proxy/rewrite rules.
# Mirrors deploy/nginx/security-headers.conf.

<IfModule mod_headers.c>
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
  Header always set X-Frame-Options "DENY"
  Header always set X-Content-Type-Options "nosniff"
  # A client page's address (/client/<link>) is its private link: it is never sent on as a referrer
  # (as the backend's own web server does). The second line keeps the mark after the SPA rewrite.
  SetEnvIf Request_URI "^/client/" MICO360_CLIENT_PAGE
  SetEnvIf REDIRECT_MICO360_CLIENT_PAGE .+ MICO360_CLIENT_PAGE
  Header always set Referrer-Policy "strict-origin-when-cross-origin" env=!MICO360_CLIENT_PAGE
  Header always set Referrer-Policy "no-referrer" env=MICO360_CLIENT_PAGE
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=()"
  Header always set Content-Security-Policy "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'; font-src 'self' data:; img-src 'self' data: blob:; connect-src 'self' wss://task.mico360.com; frame-ancestors 'none'; base-uri 'self'; object-src 'none'; form-action 'self'"

  # Search engines may list only the public pages (the site root, sign-in, privacy, terms); every
  # other page is private. The first line marks a public address; the second keeps the mark when
  # the SPA rewrite hands the request to index.html. (The API and uploaded files get the same
  # header from the backend, and the web app also sets a robots meta tag on private pages.)
  SetEnvIf Request_URI "^/(login|privacy|terms)?/?$" MICO360_PUBLIC_PAGE
  SetEnvIf REDIRECT_MICO360_PUBLIC_PAGE .+ MICO360_PUBLIC_PAGE
  <FilesMatch "^index\.html$">
    Header set X-Robots-Tag "noindex, nofollow" env=!MICO360_PUBLIC_PAGE
  </FilesMatch>

  # index.html must always be revalidated so browsers pick up new bundle hashes after a deploy.
  <FilesMatch "^index\.html$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
  # Vite's fingerprinted bundles never change, so they can be cached for a year.
  <FilesMatch "\.(js|css|woff2?)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  # The service worker (sw.js) isn't fingerprinted: browsers must check it on each visit, so a new
  # one (or its kill switch) is picked up at once. After the rule above, so it wins.
  <FilesMatch "^sw\.js$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
  # The Android app (downloads/MICO360-Tasks.apk): the sign-in page's "Download for Android" button
  # always uses this one address, and a new build replaces the file in place. Browsers must check
  # for a new file each time, and save it rather than open it.
  <FilesMatch "^MICO360-Tasks\.apk$">
    Header set Cache-Control "no-cache"
    Header set Content-Disposition "attachment; filename=\"MICO360-Tasks.apk\""
  </FilesMatch>
</IfModule>

<IfModule mod_mime.c>
  AddType application/vnd.android.package-archive .apk
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml
</IfModule>
