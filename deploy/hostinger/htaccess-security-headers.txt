# Security headers + caching for the MICO360 Tasks web app on Hostinger (LiteSpeed reads .htaccess).
# MERGE these lines into the existing .htaccess in the web root that serves the React build —
# do not replace that file, it may already contain Hostinger's proxy/rewrite rules.
# Mirrors deploy/nginx/security-headers.conf.

<IfModule mod_headers.c>
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
  Header always set X-Frame-Options "DENY"
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=()"
  Header always set Content-Security-Policy "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' https://fonts.gstatic.com data:; img-src 'self' data: blob:; connect-src 'self' wss://task.mico360.com; frame-ancestors 'none'; base-uri 'self'; object-src 'none'; form-action 'self'"

  # index.html must always be revalidated so browsers pick up new bundle hashes after a deploy.
  <FilesMatch "^index\.html$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
  # Vite's fingerprinted bundles never change, so they can be cached for a year.
  <FilesMatch "\.(js|css|woff2?)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml
</IfModule>
